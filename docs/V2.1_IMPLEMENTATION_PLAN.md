```python
def run_synthesis_core(config: SynthesisConfig) -> SynthesisResult:
    """Core synthesis logic that returns data structures."""

    """CLI wrapper that calls core logic."""
```

```python
class Synthesizer:
    def __init__(self, base_binary: Path):
        """Initialize synthesizer with base binary."""
        self.base_binary = Path(base_binary)
        if not self.base_binary.exists():
            raise FileNotFoundError(f"Binary not found: {base_binary}")
    
    def generate(
        self,
        anchor_rva: Optional[str] = None,
        anchor_fo: Optional[str] = None,
        anchor_va: Optional[str] = None,
        profile: str = "default",
        versions: Optional[List[Path]] = None,
        explain: bool = False,
        **kwargs
    ) -> SynthesisResult:
        """
        Generate AoB signatures programmatically.
        
        This is the primary SDK entry point.
        Calls run_synthesis_core() from synth.py.
        
        Returns:
            SynthesisResult with candidates, warnings, trace (if explain=True)
        
        Raises:
            ValueError: Invalid configuration
            FileNotFoundError: Binary not found
            SynthesisError: Synthesis failed
        """
        config = SynthesisConfig(
            base_binary=self.base_binary,
            anchor_rva=anchor_rva,
            anchor_fo=anchor_fo,
            anchor_va=anchor_va,
            profile=profile,
            version_binaries=versions,
            explain=explain,
            **kwargs
        )
        
        # Call core synthesis logic
        from .synth import run_synthesis_core
        return run_synthesis_core(config)
```

- Explainability mode

```python
# Example: Automated signature rotation
def rotate_broken_signature(db: SignatureDatabase, signature_id: str):
    """
    When a signature breaks, automatically generate replacement.
    """
    sig = db.query_signature(signature_id)
    synth = Synthesizer(sig['base_binary'])
    
    # Regenerate with more aggressive wildcarding
    result = synth.generate(
        anchor_rva=sig['anchor_rva'],
        profile='aggressive',
        explain=True
    )
    
    if result.ok:
        new_sig_id = f"{signature_id}_v2"
        db.save_signature(
            signature_id=new_sig_id,
            name=f"{sig['name']} (regenerated)",
            pattern=result.get_top_pattern(),
            metadata={
                'parent_id': signature_id,
                'reason': 'automatic_rotation',
                'profile': 'aggressive'
            }
        )
        return new_sig_id
    return None

# Example: Predictive maintenance
def check_signature_health(
    db: SignatureDatabase,
    analyzer: TemporalAnalyzer
) -> List[str]:
    """
    Identify signatures that need attention.
    """
    at_risk = []
    
    for sig in db.list_signatures():
        analysis = analyzer.analyze_signature(sig['id'])
        
        if analysis['stability'] == 'fragile':
            at_risk.append(sig['id'])
    
    return at_risk
```

AoBMaster v2.1 completes the v2 vision by delivering a **production-ready, library-first SDK** with **enhanced intelligence** and **superior performance**. This release transforms AoBMaster from a CLI tool into a **platform** suitable for enterprise automation and integration.

**Key Value Propositions**:

5. **Quality**: 100% backward compatible, >90% test coverage
