## Executive Summary

AoBMaster v2.2 represents a **strategic expansion** of platform capabilities with **multi-architecture support** and **advanced structural analysis**. This release transforms AoBMaster from a PE x64-only tool into a **cross-platform signature intelligence system** capable of handling diverse binary formats.

**Key Objectives**:
1. Multi-architecture support (ARM64, x86-32, MIPS)
2. Advanced structural analysis with CFG reconstruction
3. Optional cloud signature repository (community-driven)
4. Optional web dashboard for signature browsing
5. Enhanced alignment algorithms for cross-architecture patterns

**Note**: This is a **major expansion** that significantly increases system complexity. Features are designed to be **modular and opt-in** to maintain backward compatibility.

```python
from abc import ABC, abstractmethod
from typing import List, Dict, Any
from dataclasses import dataclass

@dataclass
class Instruction:
    """Architecture-agnostic instruction representation."""
    address: int
    bytes: bytes
    mnemonic: str
    operands: List[str]
    length: int
    metadata: Dict[str, Any]  # Architecture-specific data

class ArchitectureBackend(ABC):
    """
    Abstract base class for architecture-specific implementations.
    
    Each architecture (x64, x86, ARM64, etc.) implements this interface.
    """
    
    @property
    @abstractmethod
    def name(self) -> str:
        """Architecture name (e.g., 'x86-64', 'ARM64')."""
        pass
    
    @property
    @abstractmethod
    def pointer_size(self) -> int:
        """Pointer size in bytes (4 for 32-bit, 8 for 64-bit)."""
        pass
    
    @abstractmethod
    def disassemble(
        self,
        data: bytes,
        address: int,
        count: int = 10
    ) -> List[Instruction]:
        """Disassemble instructions at address."""
        pass
    
    @abstractmethod
    def get_wildcarding_rules(self, profile: str) -> Dict[str, Any]:
        """
        Get architecture-specific wildcarding rules.
        
        Returns:
            {
                "wildcard_branches": bool,
                "wildcard_pc_relative": bool,
                "wildcard_stack_offsets": bool,
                "wildcard_immediates": bool,
                "custom_patterns": List[Dict]
            }
        """
        pass
    
    @abstractmethod
    def is_pc_relative(self, insn: Instruction) -> bool:
        """Check if instruction uses PC-relative addressing."""
        pass
    
    @abstractmethod
    def is_branch(self, insn: Instruction) -> bool:
        """Check if instruction is a branch/jump/call."""
        pass
    
    @abstractmethod
    def is_stack_access(self, insn: Instruction) -> bool:
        """Check if instruction accesses stack."""
        pass
    
    @abstractmethod
    def extract_displacement(self, insn: Instruction) -> Optional[int]:
        """Extract displacement/offset from instruction."""
        pass
    
    @abstractmethod
    def normalize_instruction(
        self,
        insn: Instruction,
        rules: Dict[str, Any]
    ) -> bytes:
        """
        Apply normalization (wildcarding) to instruction bytes.
        
        Returns normalized bytes with wildcards (0x3F for '??').
        """
        pass
    
    @abstractmethod
    def detect_function_prologue(
        self,
        data: bytes,
        address: int
    ) -> Optional[int]:
        """
        Detect function prologue pattern.
        
        Returns offset of function start, or None if not found.
        """
        pass
```

from capstone import Cs, CS_ARCH_ARM64, CS_MODE_ARM

class ARM64Backend(ArchitectureBackend):
    """ARM64 (AArch64) architecture backend using Capstone."""
    
    def __init__(self):
        self.cs = Cs(CS_ARCH_ARM64, CS_MODE_ARM)
        self.cs.detail = True
    
    @property
    def name(self) -> str:
        return "ARM64"
    
    @property
    def pointer_size(self) -> int:
        return 8
    
    def disassemble(self, data: bytes, address: int, count: int = 10) -> List[Instruction]:
        """Disassemble using Capstone."""
        instructions = []
        for insn in self.cs.disasm(data, address):
            instructions.append(Instruction(
                address=insn.address,
                bytes=insn.bytes,
                mnemonic=insn.mnemonic,
                operands=[insn.op_str],
                length=insn.size,
                metadata={'cs_insn': insn}
            ))
            if len(instructions) >= count:
                break
        return instructions
    
    def get_wildcarding_rules(self, profile: str) -> Dict[str, Any]:
        """
        ARM64 specific wildcarding rules.
        
        Key considerations:
        - ADRP + ADD/LDR pairs for PC-relative addressing
        - Branch instructions (B, BL, CBZ, etc.)
        - Stack pointer manipulations (SP-relative addressing)
        - Literal pools (LDR from PC-relative offset)
        """
        if profile == "default":
            return {
                "wildcard_branches": True,
                "wildcard_adrp_pairs": True,  # ARM64 specific
                "wildcard_sp_relative": True,
                "wildcard_literal_loads": True,
                "wildcard_immediates": False
            }
        # ... other profiles
    
    def is_pc_relative(self, insn: Instruction) -> bool:
        """
        Check for PC-relative addressing.
        
        ARM64 PC-relative patterns:
        - ADRP (address of page, PC-relative)
        - ADR (address, PC-relative)
        - LDR with PC-relative offset
        """
        cs_insn = insn.metadata.get('cs_insn')
        if not cs_insn:
            return False
        
        # Check for ADRP, ADR, or PC-relative LDR
        if cs_insn.mnemonic in ['adrp', 'adr']:
            return True
        
        if cs_insn.mnemonic.startswith('ldr'):
            # Check if any operand references PC
            for op in cs_insn.operands:
                if op.type == ARM64_OP_MEM and op.mem.base == ARM64_REG_PC:
                    return True
        
        return False
    
    def is_branch(self, insn: Instruction) -> bool:
        """
        Check for branch instructions.
        
        ARM64 branches:
        - B (unconditional branch)
        - BL (branch with link - call)
        - BR (branch to register)
        - BLR (branch with link to register)
        - CBZ, CBNZ (compare and branch)
        - TBZ, TBNZ (test bit and branch)
        - Conditional branches (B.EQ, B.NE, etc.)
        """
        cs_insn = insn.metadata.get('cs_insn')
        if not cs_insn:
            return False
        
        return cs_insn.mnemonic in [
            'b', 'bl', 'br', 'blr',
            'cbz', 'cbnz', 'tbz', 'tbnz',
            'b.eq', 'b.ne', 'b.cs', 'b.cc', 'b.mi', 'b.pl',
            'b.vs', 'b.vc', 'b.hi', 'b.ls', 'b.ge', 'b.lt',
            'b.gt', 'b.le', 'b.al'
        ]
    
    def normalize_instruction(self, insn: Instruction, rules: Dict) -> bytes:
        """
        Apply ARM64-specific normalization.
        
        Wildcarding targets:
        1. Branch offsets (B, BL - immediate offsets)
        2. ADRP page addresses
        3. SP-relative offsets (LDR [SP, #imm])
        4. Literal pool loads (LDR from PC-relative)
        5. Optionally: immediate values
        """
        normalized = bytearray(insn.bytes)
        cs_insn = insn.metadata.get('cs_insn')
        
        if rules.get('wildcard_branches') and self.is_branch(insn):
            # Wildcard branch offset (varies by instruction)
            if cs_insn.mnemonic in ['b', 'bl']:
                # B/BL: 26-bit signed offset in bits [25:0]
                # Wildcard bytes 0-2 (offset is in these bytes)
                normalized[0] = 0x3F  # Wildcard marker
                normalized[1] = 0x3F
                normalized[2] = 0x3F
        
        if rules.get('wildcard_adrp_pairs') and cs_insn.mnemonic == 'adrp':
            # ADRP: 21-bit page offset in bits [23:5] and [30:29]
            # Wildcard page address bytes
            normalized[1] = 0x3F
            normalized[2] = 0x3F
        
        if rules.get('wildcard_sp_relative') and self.is_stack_access(insn):
            # SP-relative loads/stores: wildcard offset
            # Pattern varies by instruction (LDR, STR, STP, LDP)
            # Implementation details...
            pass
        
        return bytes(normalized)
    
    def detect_function_prologue(self, data: bytes, address: int) -> Optional[int]:
        """
        Detect ARM64 function prologue patterns.
        
        Common patterns:
        - STP X29, X30, [SP, #-<imm>]!  (save FP and LR)
        - SUB SP, SP, #<imm>  (allocate stack frame)
        - MOV X29, SP  (set frame pointer)
        """
        instructions = self.disassemble(data, address, count=10)
        
        for i, insn in enumerate(instructions):
            cs_insn = insn.metadata.get('cs_insn')
            
            # Look for STP X29, X30 (frame pointer + link register save)
            if cs_insn.mnemonic == 'stp':
                ops = cs_insn.op_str.lower()
                if 'x29' in ops and 'x30' in ops and 'sp' in ops:
                    return insn.address
            
            # Look for SUB SP, SP at function start
            if cs_insn.mnemonic == 'sub':
                ops = cs_insn.op_str.lower()
                if ops.startswith('sp, sp'):
                    return insn.address
        
        return None

from typing import Dict, Type
from .arch import ArchitectureBackend

class ArchitectureRegistry:
    """
    Registry for architecture backends.
    
    Allows plugins to register new architectures dynamically.
    """
    
    _backends: Dict[BinaryFormat, Type[ArchitectureBackend]] = {}
    
    @classmethod
    def register(
        cls,
        format: BinaryFormat,
        backend: Type[ArchitectureBackend]
    ) -> None:
        """Register architecture backend."""
        cls._backends[format] = backend
    
    @classmethod
    def get_backend(cls, format: BinaryFormat) -> ArchitectureBackend:
        """Get backend for binary format."""
        if format not in cls._backends:
            raise ValueError(f"Unsupported binary format: {format}")
        return cls._backends[format]()
    
    @classmethod
    def list_supported(cls) -> List[BinaryFormat]:
        """List all supported binary formats."""
        return list(cls._backends.keys())

##### 2.1 Control Flow Graph (CFG) Reconstruction
**File**: `aobmaster/cfg.py` (NEW - ~500 lines)

```python
from dataclasses import dataclass
from typing import List, Dict, Set, Optional

@dataclass
class BasicBlock:
    """
    Basic block in control flow graph.
    
    A basic block is a straight-line sequence of instructions
    with no branches except at the end.
    """
    start_address: int
    end_address: int
    instructions: List[Instruction]
    successors: List[int]  # Addresses of successor blocks
    predecessors: List[int]  # Addresses of predecessor blocks
    block_type: str  # "entry", "exit", "normal", "return"

@dataclass
class Function:
    """Function representation with CFG."""
    start_address: int
    end_address: Optional[int]
    name: Optional[str]
    basic_blocks: Dict[int, BasicBlock]  # address -> block
    entry_block: int
    exit_blocks: List[int]
    call_targets: List[int]  # Functions called by this function
    is_leaf: bool  # No function calls

class CFGBuilder:
    """
    Build control flow graph from disassembled code.
    
    Uses recursive traversal starting from function entry points.
    """
    
    def __init__(self, arch: ArchitectureBackend):
        self.arch = arch
    
    def build_function_cfg(
        self,
        data: bytes,
        entry_address: int,
        max_instructions: int = 1000
    ) -> Function:
        """
        Build CFG for a single function.
        
        Algorithm:
        1. Start at entry address
        2. Disassemble instructions sequentially
        3. Detect basic block boundaries:
           - Branch target addresses
           - Instructions following branches
           - Return instructions
        4. Build edges between basic blocks
        5. Detect function exit (return instructions)
        
        Returns:
            Function object with complete CFG
        """
        visited = set()
        basic_blocks = {}
        work_queue = [entry_address]
        
        while work_queue:
            address = work_queue.pop(0)
            if address in visited:
                continue
            
            # Build basic block starting at address
            block = self._build_basic_block(data, address)
            basic_blocks[address] = block
            visited.add(address)
            
            # Add successor blocks to work queue
            for succ in block.successors:
                if succ not in visited:
                    work_queue.append(succ)
        
        return Function(
            start_address=entry_address,
            end_address=self._find_function_end(basic_blocks),
            name=None,
            basic_blocks=basic_blocks,
            entry_block=entry_address,
            exit_blocks=self._find_exit_blocks(basic_blocks),
            call_targets=self._extract_call_targets(basic_blocks),
            is_leaf=self._is_leaf_function(basic_blocks)
        )
    
    def _build_basic_block(
        self,
        data: bytes,
        start_address: int
    ) -> BasicBlock:
        """Build a single basic block."""
        instructions = []
        address = start_address
        
        while True:
            insn = self.arch.disassemble(data, address, count=1)[0]
            instructions.append(insn)
            address += insn.length
            
            # Stop at branch, return, or invalid instruction
            if self.arch.is_branch(insn) or self._is_return(insn):
                break
            
            # Stop if we hit another function (heuristic)
            if self.arch.detect_function_prologue(data, address):
                break
        
        # Determine successors
        last_insn = instructions[-1]
        successors = self._get_successors(last_insn, address)
        
        return BasicBlock(
            start_address=start_address,
            end_address=address,
            instructions=instructions,
            successors=successors,
            predecessors=[],  # Filled in later
            block_type=self._classify_block(instructions)
        )
    
    def _get_successors(
        self,
        insn: Instruction,
        next_address: int
    ) -> List[int]:
        """
        Determine successor addresses for instruction.
        
        Cases:
        - Unconditional branch: [target]
        - Conditional branch: [target, next_address]
        - Return: []
        - Call: [next_address]
        - Normal: [next_address]
        """
        if self._is_return(insn):
            return []
        
        if self.arch.is_branch(insn):
            target = self._extract_branch_target(insn)
            
            if self._is_unconditional_branch(insn):
                return [target] if target else []
            else:
                # Conditional branch: both taken and not-taken
                return [target, next_address] if target else [next_address]
        
        # Normal instruction: fall through
        return [next_address]
    
    def _is_return(self, insn: Instruction) -> bool:
        """Check if instruction is a return."""
        # Architecture-specific
        if self.arch.name == "x86-64":
            return insn.mnemonic == "ret"
        elif self.arch.name == "ARM64":
            return insn.mnemonic == "ret"
        elif self.arch.name == "MIPS64":
            return insn.mnemonic in ["jr $ra", "j $ra"]
        return False
```

##### 2.2 Call-Site Anchoring
**File**: `aobmaster/call_site_anchor.py` (NEW - ~250 lines)

```python
class CallSiteAnchor:
    """
    Advanced anchoring based on function call sites.
    
    Instead of anchoring on arbitrary instruction, anchor on:
    - Function calls (stable across versions)
    - Function returns
    - Known library calls (e.g., malloc, free, printf)
    
    More stable than instruction-level anchoring.
    """
    
    def __init__(self, cfg: Function):
        self.cfg = cfg
    
    def find_call_site_anchors(self) -> List[Dict[str, Any]]:
        """
        Find potential call-site anchors in function.
        
        Returns:
            [
                {
                    "address": int,
                    "type": "call" | "return",
                    "target": int | None,  # For calls
                    "stability_score": float,  # 0.0-1.0
                    "reason": str
                }
            ]
        """
        anchors = []
        
        for block in self.cfg.basic_blocks.values():
            for insn in block.instructions:
                if self._is_call_instruction(insn):
                    target = self._extract_call_target(insn)
                    
                    anchors.append({
                        "address": insn.address,
                        "type": "call",
                        "target": target,
                        "stability_score": self._score_call_stability(insn, target),
                        "reason": self._explain_call_stability(insn, target)
                    })
                
                elif self._is_return_instruction(insn):
                    anchors.append({
                        "address": insn.address,
                        "type": "return",
                        "target": None,
                        "stability_score": 0.8,  # Returns are fairly stable
                        "reason": "Function return point"
                    })
        
        return sorted(anchors, key=lambda x: x['stability_score'], reverse=True)
    
    def _score_call_stability(
        self,
        insn: Instruction,
        target: Optional[int]
    ) -> float:
        """
        Score call site stability.
        
        Higher score = more stable across versions
        
        Factors:
        - Direct call vs indirect: Direct is more stable (0.9 vs 0.5)
        - Library call vs internal: Library is more stable (0.95 vs 0.7)
        - Distance from function entry: Closer = more stable
        - Block type: Entry blocks more stable than loop blocks
        """
        score = 0.5  # Base score
        
        if target:
            # Direct call (vs indirect through register)
            score += 0.3
            
            # Check if target looks like library function
            if self._is_likely_library_function(target):
                score += 0.2
        
        # Proximity to function entry
        distance_from_entry = insn.address - self.cfg.start_address
        if distance_from_entry < 50:
            score += 0.1
        
        return min(score, 1.0)
```
